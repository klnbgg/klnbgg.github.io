2.2Delay 延时函数模块P4 [2-2]1-3 System/Delay.c

利用 Cortex-M3 内核的 SysTick 系统滴答定时器（24位递减计数器）实现精确微秒/毫秒延时，不占用外设。

SysTick 时钟 = HCLK/8 = 72M/8 = 9MHz（每个计数 1/9μs）；LOAD = 9 × xus 即延时 xus
System/Delay.c — SysTick 寄存器级配置（完整）
void Delay_us(uint32_t xus)
{
    uint32_t i;
    SysTick->LOAD = 72 * xus;          //重装值：9MHz下9*xus，但库默认72MHz不分频→72*xus
    SysTick->VAL = 0x00;               //清空当前值
    SysTick->CTRL = 0x00000005;        //使能SysTick，时钟源HCLK(72M)，不开中断
    for (i = 0; i < xus; i++) {
        while(!(SysTick->CTRL & 0x00010000));  //等待COUNTFLAG=1（计数到0）
    }
    SysTick->CTRL = 0x00000004;        //关闭SysTick
}

void Delay_ms(uint32_t xms)
{
    while(xms--) Delay_us(1000);    //1ms = 1000us
}

    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;

void OLED_I2C_Start(void) {
    OLED_W_SDA(1); OLED_W_SCL(1); OLED_W_SDA(0); OLED_W_SCL(0);
}
/*停止条件：SCL高时SDA低→高*/
void OLED_I2C_Stop(void) {
    OLED_W_SDA(0); OLED_W_SCL(1); OLED_W_SDA(1);
}
/*发送字节：高位先行，8个时钟后额外一个时钟跳过应答*/
void OLED_I2C_SendByte(uint8_t Byte) {
    for (uint8_t i=0; i<8; i++) {
        OLED_W_SDA(!!(Byte & (0x80 >> i)));
        OLED_W_SCL(1); OLED_W_SCL(0);
    }
    OLED_W_SCL(1); OLED_W_SCL(0);  //额外时钟，不处理应答
}

void OLED_WriteCommand(uint8_t Command) {

EXTI 外部中断

中断原理、NVIC 优先级、EXTI 外部中断及两个实验
5.1EXTI 外部中断P11 [5-1]

中断让 CPU 暂停当前任务，优先处理紧急事件。

主程序 → 中断触发 → 保存断点 → 中断服务函数 → 恢复断点 → 继续主程序
  • NVIC 优先级：抢占优先级（可嵌套）+ 响应优先级（排队不可嵌套），分组2：2位抢占+2位响应
  • EXTI 线路：引脚电平变化 → AFIO 选择引脚 → EXTI 边沿检测 → NVIC → 中断服务函数
  • 相同 Pin 号不能同时挂中断（PA0 和 PB0 不能同时用 EXTI0）
进入中断后必须清除中断标志位，否则反复进入中断导致主程序卡死。中断函数名必须与启动文件向量表一致。

6.1TIM 定时中断（理论）P13 [6-1]

定时器时基单元：PSC 预分频器 + CNT 计数器 + ARR 自动重装器。

CK_PSC(72MHz) → [PSC+1分频] → CK_CNT → [CNT计数] → 达到ARR → 更新事件 → 中断

  • APB1 定时器时钟经倍频后为 72MHz（APB1 预分频≠1 时×2）
TIM_TimeBaseInit() 末尾会手动产生一次更新事件，开中断前必须先 TIM_ClearFlag()。

6.4TIM 输出比较（理论）P15 [6-3]

两线制同步半双工总线：SCL 时钟线 + SDA 数据线，开漏输出+上拉电阻，支持多设备。

起始(SCL高时SDA高→低) → 从机地址7位+读写位 → 应答 → 寄存器地址 → 应答 → 数据 → 应答 → 停止(SCL高时SDA低→高)

  • 6 轴数据：加速度 X/Y/Z + 陀螺仪 X/Y/Z，每轴 16 位（H/L 两寄存器拼接）

void MyI2C_W_SCL(uint8_t BitValue) {

    Delay_us(10);
}
void MyI2C_W_SDA(uint8_t BitValue) {

    Delay_us(10);
}

    uint8_t BitValue = GPIO_ReadInputDataBit(GPIOB, GPIO_Pin_11);

    return BitValue;

    /*GPIO初始化*/

    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOB, &GPIO_InitStructure);

void MyI2C_Start(void) {
    MyI2C_W_SDA(1); MyI2C_W_SCL(1); MyI2C_W_SDA(0); MyI2C_W_SCL(0);
}
void MyI2C_Stop(void) {
    MyI2C_W_SDA(0); MyI2C_W_SCL(1); MyI2C_W_SDA(1);
}
void MyI2C_SendByte(uint8_t Byte) {
    for (i=0; i<8; i++) {
        MyI2C_W_SDA(!!(Byte & (0x80 >> i)));  //高位先行
        MyI2C_W_SCL(1); MyI2C_W_SCL(0);
    }
}
uint8_t MyI2C_ReceiveByte(void) {
    uint8_t Byte = 0;
    MyI2C_W_SDA(1);  //释放SDA
    for (i=0; i<8; i++) {
        MyI2C_W_SCL(1);
        if (MyI2C_R_SDA()) Byte |= (0x80 >> i);
        MyI2C_W_SCL(0);
    }
    return Byte;
}
void MyI2C_SendAck(uint8_t AckBit) {
    MyI2C_W_SDA(AckBit); MyI2C_W_SCL(1); MyI2C_W_SCL(0);
}
uint8_t MyI2C_ReceiveAck(void) {
    uint8_t AckBit;
    MyI2C_W_SDA(1); MyI2C_W_SCL(1);
    AckBit = MyI2C_R_SDA();
    MyI2C_W_SCL(0);
    return AckBit;
}

/*写寄存器：START→EV5→地址Tx→EV6→Reg→EV8→Data→EV8_2→STOP*/
void MPU6050_WriteReg(uint8_t RegAddress, uint8_t Data) {
    I2C_GenerateSTART(I2C2, ENABLE);
    MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_MODE_SELECT);  //EV5
    I2C_Send7bitAddress(I2C2, 0xD0, I2C_Direction_Transmitter);
    MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED);  //EV6
    I2C_SendData(I2C2, RegAddress);
    MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_BYTE_TRANSMITTING);  //EV8
    I2C_SendData(I2C2, Data);
    MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_BYTE_TRANSMITTED);  //EV8_2
    I2C_GenerateSTOP(I2C2, ENABLE);
}

/*读寄存器：写完Reg后重复START，最后一字节前提前NACK+STOP*/
uint8_t MPU6050_ReadReg(uint8_t RegAddress) {
    I2C_GenerateSTART(I2C2, ENABLE); MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_MODE_SELECT);
    I2C_Send7bitAddress(I2C2, 0xD0, I2C_Direction_Transmitter); MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED);
    I2C_SendData(I2C2, RegAddress); MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_BYTE_TRANSMITTED);
    I2C_GenerateSTART(I2C2, ENABLE); MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_MODE_SELECT);
    I2C_Send7bitAddress(I2C2, 0xD0, I2C_Direction_Receiver); MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_RECEIVER_MODE_SELECTED);
    I2C_AcknowledgeConfig(I2C2, DISABLE);  //最后一字节前提前NACK
    I2C_GenerateSTOP(I2C2, ENABLE);
    MPU6050_WaitEvent(I2C2, I2C_EVENT_MASTER_BYTE_RECEIVED);  //EV7
    Data = I2C_ReceiveData(I2C2);
    I2C_AcknowledgeConfig(I2C2, ENABLE);  //恢复应答
    return Data;
}

System/MyRTC.c — 完整初始化+时间读写
void MyRTC_Init(void)
{
    /*开启时钟*/
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_PWR | RCC_APB1Periph_BKP, ENABLE);
    /*备份寄存器访问使能*/
    PWR_BackupAccessCmd(ENABLE);

    /*BKP_DR1不是0xA5A5 → 首次上电，需要配置RTC*/
    if (BKP_ReadBackupRegister(BKP_DR1) != 0xA5A5)
    {
        RCC_LSEConfig(RCC_LSE_ON);
        while (RCC_GetFlagStatus(RCC_FLAG_LSERDY) == RESET);  //等LSE就绪
        RCC_RTCCLKConfig(RCC_RTCCLKSource_LSE);
        RCC_RTCCLKCmd(ENABLE);
        RTC_WaitForSynchro();
        RTC_WaitForLastTask();
        RTC_SetPrescaler(32768 - 1);  //32768Hz → 1Hz
        RTC_WaitForLastTask();
        MyRTC_SetTime(2024, 1, 1, 0, 0, 0);  //初始时间
        BKP_WriteBackupRegister(BKP_DR1, 0xA5A5);  //写标志
    }
    else {
        RTC_WaitForSynchro();
        RTC_WaitForLastTask();
    }
}

/*时间→时间戳（减8小时转UTC）→写RTC计数器*/
void MyRTC_SetTime(uint16_t Year, uint8_t Month, uint8_t Day,
                   uint8_t Hour, uint8_t Minute, uint8_t Second)
{
    struct tm t;
    t.tm_year = Year - 1900; t.tm_mon = Month - 1; t.tm_mday = Day;
    t.tm_hour = Hour; t.tm_min = Minute; t.tm_sec = Second;
    uint32_t Timestamp = mktime(&t) - 8 * 3600;  //北京时间→UTC
    RTC_WaitForLastTask();
    RTC_SetCounter(Timestamp);
    RTC_WaitForLastTask();
}

/*读RTC计数器→加8小时→localtime转北京时间*/
void MyRTC_ReadTime(...) {
    uint32_t Timestamp = RTC_GetCounter() + 8 * 3600;
    struct tm *t = localtime(&Timestamp);
    *Year = t->tm_year + 1900; *Month = t->tm_mon + 1; ...
}

14.1独立看门狗 IWDGP45-P46 [14-1][14-2]14-1 User/main.c

LSI 约 40kHz 独立时钟驱动，超时 1 秒。先检查复位标志判断是否看门狗复位，正常循环中喂狗，按键阻塞模拟卡死时不喂狗→复位。

    OLED_ShowString(1, 1, "IWDG RST");
    RCC_ClearFlag();

IWDG_SetReload(2499);                          //2.5kHz/2500=1秒

while(1) {

15.1读写内部 FLASH（理论）P47 [15-1]

STM32F103C8T6 主 Flash 64KB，每页 1KB。写入前必须擦除（擦除后全 0xFF），只能 1→0。

  • 解锁序列：FLASH_Unlock() → 操作 → FLASH_Lock()
  • 编程必须按半字（16位）写入；读可以按字节/半字/字用指针直接读
页1KB先擦后写Unlock/Lock半字编程指针读取

15.3Store 参数存储模块P48 [15-2]15-1 System/Store.c

用最后一页（0x0800FC00，1KB）存储参数，SRAM 数组做缓存。第一个半字写 0xA5A5 标志判断首次使用，上电加载 Flash→SRAM，保存时擦除后全部写回。

System/Store.c — 完整初始化+保存+清零
#define STORE_START_ADDRESS  0x0800FC00  //最后一页（64KB-1KB）
#define STORE_COUNT         512         //1KB / 2字节 = 512个半字
uint16_t Store_Data[STORE_COUNT];       //SRAM缓存数组

void Store_Init(void)
{
    /*首次使用：标志位不是0xA5A5 → 擦除→写标志→其余清零*/
    if (MyFLASH_ReadHalfWord(STORE_START_ADDRESS) != 0xA5A5)
    {
        MyFLASH_ErasePage(STORE_START_ADDRESS);
        MyFLASH_ProgramHalfWord(STORE_START_ADDRESS, 0xA5A5);
        for (i=1; i<STORE_COUNT; i++)
            MyFLASH_ProgramHalfWord(STORE_START_ADDRESS + i*2, 0x0000);
    }
    /*加载Flash数据到SRAM数组*/
    for (i=0; i<STORE_COUNT; i++)
        Store_Data[i] = MyFLASH_ReadHalfWord(STORE_START_ADDRESS + i*2);
}

void Store_Save(void)
{
    MyFLASH_ErasePage(STORE_START_ADDRESS);
    for (i=0; i<STORE_COUNT; i++)
        MyFLASH_ProgramHalfWord(STORE_START_ADDRESS + i*2, Store_Data[i]);
}

void Store_Clear(void)
{
    for (i=1; i<STORE_COUNT; i++) Store_Data[i] = 0x0000;
    Store_Save();
}
